[Incident Queue · Java]

package dataStructuresAssessment;

import java.util.ArrayDeque;
import java.util.Deque;
import java.util.Iterator;

//IncidentQueue class: A queue class that stores all instances of the Incident class in order of priority.
public class IncidentQueue {
	
	private Deque incidents;
	
	//Create the incident queue
	public IncidentQueue() {
		incidents = new ArrayDeque<>();
	}
	
	//Adds an incident to the bottom (tail) of the queue
	public void lowPriorityAdd (Incident x) {
		incidents.addLast(x);
	}
	
	//Adds an incident to the top (head) of the queue
	public void highPriorityAdd (Incident x) {
		incidents.addFirst(x);
	}
	
	//Prints but does not remove the details of the incident at the top of the queue.
	public void peek() {
		System.out.println(incidents.element().getIncidentNum() + ", " + incidents.element().getDesc() + ", " + incidents.element().getService() + ", " + incidents.element().getDistrict() + ", High Priority: " + incidents.element().isHighPriority());
	}
	
	//Removes the top of the queue
	public void handleTop() {
		incidents.pollFirst();
	}
	
	//Iterates and prints every incident in the queue
	public void listAll() {
		Iterator i = incidents.iterator();
		while (i.hasNext()) {
			Incident x = i.next();
			System.out.println(x.getIncidentNum() + ", " + x.getDesc() + ", " + x.getService() + ", " + x.getDistrict() + ", High Priority: " + x.isHighPriority());
		}
	}
	
	//Checks if the queue has Incidents and returns a boolean accordingly
	public boolean tryPeek() {
		if (incidents.peekFirst() == null) {
			return false;
		} else {
			return true;
		}
	}
	
	//Prints all incidents within the user-entered district
	public String searchDistrict(String x) {
		String ret = "";
		Iterator i = incidents.iterator();
		while (i.hasNext()) {
			Incident b = i.next();
			if (b.getDistrict().equals(x)) {
				ret = ret.concat(b.getIncidentNum() + ", " + b.getDesc() + ", " + b.getService() + ", " + b.getDistrict() + ", High Priority: " + b.isHighPriority() + "\n");
			}
		} if (ret.equals("")) {
			ret = "-1";
		}
		return ret;
	}
	
	//Prints all incidents that requires the user-entered service
	public String searchService(String x) {
		String ret = "";
		Iterator i = incidents.iterator();
		while (i.hasNext()) {
			Incident b = i.next();
			if (b.getService().equals(x)) {
				ret = ret.concat(b.getIncidentNum() + ", " + b.getDesc() + ", " + b.getService() + ", " + b.getDistrict() + ", High Priority: " + b.isHighPriority() + "\n");
			}
		} if (ret.equals("")) {
			ret = "-1";
		}
		return ret;
	}
	
}
